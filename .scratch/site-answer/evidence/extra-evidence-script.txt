// The fill and the size of each tile, icon box, row, and button of the Site answers in the Thread.
const MEASURE = () => {
  const box = (el) => { const r = el.getBoundingClientRect(); return Math.round(r.height * 10) / 10; };
  const bg = (el) => getComputedStyle(el).backgroundColor;
  const out = [];
  document.querySelectorAll("article").forEach((art) => {
    const q = art.getAttribute("aria-label");
    art.querySelectorAll("ul[aria-labelledby] > li > a").forEach((a) => {
      const group = document.getElementById(a.closest("ul").getAttribute("aria-labelledby")).innerText;
      const icon = a.querySelector("span > span.grid");
      const label = a.querySelector("span.font-bold")?.innerText;
      out.push({ answer: q, kind: "tile", group, label, height: box(a), tile_bg: bg(a), icon_bg: icon ? bg(icon) : null });
    });
    art.querySelectorAll('ul[aria-label="Pages"] > li > a').forEach((a) => {
      const icon = a.querySelector("span.grid");
      out.push({ answer: q, kind: "row", label: a.querySelector("span.font-semibold")?.innerText, height: box(a), row_bg: bg(a), icon_bg: icon ? bg(icon) : null });
    });
    art.querySelectorAll("a").forEach((a) => {
      if (!a.innerText.startsWith("Open the")) return;
      const card = a.parentElement.querySelector("span.grid");
      let up = a.parentElement; while (up && /rgba\(0, 0, 0, 0\)/.test(bg(up))) up = up.parentElement;
      out.push({ answer: q, kind: "open button", label: a.innerText.trim(), height: box(a), button_bg: bg(a), card_icon_bg: card ? bg(card) : null, card_bg: up ? bg(up) : null });
    });
    art.querySelectorAll("button").forEach((b) => {
      if (!b.innerText.startsWith("All pages")) return;
      out.push({ answer: q, kind: "fold button", label: b.innerText.trim(), height: box(b), button_bg: bg(b) });
    });
  });
  const surface = (sel) => { const el = document.querySelector(sel); return el ? bg(el) : null; };
  const main = document.querySelector("main");
  let p = main; while (p && /rgba\(0, 0, 0, 0\)/.test(bg(p))) p = p.parentElement;
  return { body_bg: bg(document.body), page_bg: p ? bg(p) : null, rows: out };
};

#### serve.mjs
// Serves a static export on 127.0.0.1:4173. A missing path gets 404.html with the status 404, as the Worker gives it.
import { createServer } from "node:http";
import { readFile, stat } from "node:fs/promises";
import { join, extname } from "node:path";
const root = process.argv[2];
const types = { ".html": "text/html", ".js": "text/javascript", ".css": "text/css", ".json": "application/json", ".png": "image/png", ".jpg": "image/jpeg", ".svg": "image/svg+xml", ".txt": "text/plain", ".woff2": "font/woff2", ".webp": "image/webp", ".ico": "image/x-icon" };
async function find(path) {
  for (const p of [path, join(path, "index.html"), path + ".html"]) {
    try { if ((await stat(p)).isFile()) return p; } catch {}
  }
  return null;
}
createServer(async (req, res) => {
  const url = decodeURIComponent(new URL(req.url, "http://x").pathname);
  const file = await find(join(root, url));
  if (!file) { res.writeHead(404, { "content-type": "text/html" }); res.end(await readFile(join(root, "404.html"))); return; }
  res.writeHead(200, { "content-type": types[extname(file)] ?? "application/octet-stream" });
  res.end(await readFile(file));
}).listen(4173, "127.0.0.1", () => console.log("serving", root));

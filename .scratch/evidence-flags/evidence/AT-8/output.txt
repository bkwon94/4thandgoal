# The script, labels.py:
"""Count the labels of the evidence-flags label file, and search it for each manager name.
A manager name is the team name and the owner of each entry of clients/cpc/build/2026/managers.json."""
import json
from collections import Counter
from pathlib import Path

labels_path = Path("apps/pipeline/src/cornerstone/evidence_labels.json")
labels = json.loads(labels_path.read_text())
text = labels_path.read_text().lower()
print(f"tests: {len(labels['tests'])}")
for label, count in sorted(Counter(t["label"] for t in labels["tests"]).items()):
    print(f"{label}: {count}")
managers = json.loads(Path("clients/cpc/build/2026/managers.json").read_text())["managers"]
names = [n for m in managers for n in (m["team"]["name"], m["team"]["owner"])]
found = [n for n in names if n.lower() in text]
print(f"manager names searched: {len(names)}")
for n in names:
    print(f"  {n!r}: {'found' if n.lower() in text else 'not found'}")
print(f"manager names found: {len(found)} {found}")

/* The latest week of the export. */
function latestWeek(): { route: string; scene: HeroScene } {
  const seasons = readdirSync(OUT).filter((name) => /^\d{4}$/.test(name)).sort().reverse();
  for (const season of seasons) {
    const weeks = readdirSync(join(OUT, season)).filter((name) => /^week-\d{2}$/.test(name)).sort();
    const week = weeks.at(-1);
    if (week) {
      const page = JSON.parse(readFileSync(join(OUT, season, week, "page.json"), "utf8"));
      return { route: `/${season}/${week}/`, scene: matchupScene(page) };
    }
  }
  throw new Error("no week");
}

/* A fresh context with a viewport, a scale, and a motion setting. Each context has empty storage, so no hint flag. */
async function fresh(
  browser: import("@playwright/test").Browser,
  width: number,
  opts: { scale?: number; reduced?: boolean; touch?: boolean; height?: number } = {},
): Promise<{ context: BrowserContext; page: Page }> {
  const context = await browser.newContext({
    viewport: { width, height: opts.height ?? 900 },
    deviceScaleFactor: opts.scale ?? 1,
    reducedMotion: opts.reduced ? "reduce" : "no-preference",
    hasTouch: opts.touch ?? false,
    isMobile: opts.touch ?? false,
  });
  return { context, page: await context.newPage() };
}

/* Opens a route in a theme. The page records each state change of each canvas, and counts animation frames. */
async function open(page: Page, route: string, theme: Theme): Promise<void> {
  await page.addInitScript((value) => localStorage.setItem("cpc.theme", value), theme);
  await page.addInitScript(() => {
    const w = window as unknown as Record<string, unknown>;
    const states: { state: string; at: number }[] = [];
    w.sceneStates = states;
    w.rafCount = 0;
    const request = window.requestAnimationFrame.bind(window);
    window.requestAnimationFrame = (callback) => {
      (w.rafCount as number)++;
      return request(callback);
    };
    new MutationObserver((changes) => {
      for (const change of changes) {
        const state = (change.target as HTMLElement).dataset.sceneState;
        if (state && states.at(-1)?.state !== state) states.push({ state, at: performance.now() });
      }
    }).observe(document, { subtree: true, attributeFilter: ["data-scene-state"] });
  });
  await page.goto(BASE + route);
  await page.waitForLoadState("networkidle");
}

const statesOf = (page: Page) =>
  page.evaluate(() => (window as unknown as { sceneStates: { state: string; at: number }[] }).sceneStates);

/* The time of the n-th entry (from 0) into a state. */
async function entryOf(page: Page, state: string, nth = 0): Promise<number> {
  await expect.poll(async () => (await statesOf(page)).filter((s) => s.state === state).length, { timeout: 15000 }).toBeGreaterThan(nth);
  return (await statesOf(page)).filter((s) => s.state === state)[nth].at;
}
const until = (page: Page, at: number) => page.waitForFunction((time) => performance.now() >= time, at, { timeout: 30000 });

/* Waits for 2.5 s after the first settle, the time of a scene screenshot. */
async function settled(page: Page): Promise<void> {
  const entry = await entryOf(page, "settle");
  await until(page, entry + 2500);
  await expect(page.locator(CANVAS).first()).toHaveAttribute("data-scene-state", "rest", { timeout: 5000 });
}

/* Waits for a settle scene to rest 2.5 s after its entry, or for a drift scene to run 2.5 s. */
async function sceneReady(page: Page): Promise<void> {
  await expect.poll(async () => (await statesOf(page)).some((s) => s.state === "settle" || s.state === "drift"), { timeout: 15000 }).toBe(true);
  const first = (await statesOf(page)).find((s) => s.state === "settle" || s.state === "drift")!;
  if (first.state === "settle") await settled(page);
  else await until(page, first.at + 2500);
}

/* The art of a hero: its label, the art layers, the count of scene canvases in it, and the count on the page. */
async function heroArt(hero: import("@playwright/test").Locator) {
  return hero.evaluate((section) => ({
    heroLabel: section.getAttribute("aria-label"),
    texture: section.querySelectorAll('[data-hero-art="texture"]').length,
    brackets: section.querySelectorAll('[data-hero-art="bracket"]').length,
    textureBackground: getComputedStyle(section.querySelector('[data-hero-art="texture"]') ?? section).backgroundImage.slice(0, 160),
    canvasesInHero: section.querySelectorAll("canvas[data-hero-scene]").length,
    canvasesOnPage: document.querySelectorAll("canvas[data-hero-scene]").length,
  }));
}

/* One frame of the canvas, drawn on the band color of the hero, written as a PNG. The capture runs in the page at once. */
async function frame(page: Page, path: string, canvas = page.locator(CANVAS).first()): Promise<{ at: number; state: string; data: string }> {
  const result = await canvas.evaluate((element: HTMLCanvasElement) => {
    const at = performance.now();
    const hero = element.closest("section") as HTMLElement;
    const copy = document.createElement("canvas");
    copy.width = element.width;
    copy.height = element.height;
    const ctx = copy.getContext("2d")!;
    ctx.fillStyle = getComputedStyle(hero).backgroundColor;
    ctx.fillRect(0, 0, copy.width, copy.height);
    ctx.drawImage(element, 0, 0);
    return { at, state: element.dataset.sceneState ?? "", data: element.toDataURL(), png: copy.toDataURL("image/png") };
  });
  writeFileSync(path, Buffer.from(result.png.split(",")[1], "base64"));
  return { at: result.at, state: result.state, data: result.data };
}

const sceneBox = (page: Page) => page.locator(CANVAS).first().locator("xpath=..");

async function lightsOf(page: Page, scene: HeroScene): Promise<ScenePoint[]> {
  const size = await page.locator(CANVAS).first().evaluate((c) => ({ width: c.clientWidth, height: c.clientHeight }));
  return placeLights(scene, { ...size, padding: SCENE_PADDING });
}
async function onPage(page: Page, point: ScenePoint): Promise<ScenePoint> {
  const box = (await page.locator(CANVAS).first().boundingBox())!;
  return { x: box.x + point.x, y: box.y + point.y };
}
async function pixelAt(page: Page, point: ScenePoint): Promise<string> {
  return page.locator(CANVAS).first().evaluate((canvas: HTMLCanvasElement, at) => {
    const ratio = canvas.width / canvas.clientWidth;
    const [r, g, b] = canvas.getContext("2d")!.getImageData(Math.round(at.x * ratio), Math.round(at.y * ratio), 1, 1).data;
    return `#${[r, g, b].map((v) => v.toString(16).padStart(2, "0")).join("")}`;
  }, point);
}
const token = (page: Page, name: string) =>
  page.evaluate((t) => getComputedStyle(document.documentElement).getPropertyValue(t).trim(), name);
const attrs = (page: Page) =>
  page.locator(CANVAS).first().evaluate((c) => ({ state: c.dataset.sceneState, label: c.dataset.sceneLabel, hint: c.dataset.sceneHint }));

/* Shots of the first hero of a route at each width and theme, after the settle when the page has a scene. */
async function heroShots(
  browser: import("@playwright/test").Browser,
  id: string,
  route: string,
  widths: number[],
  themes: readonly Theme[],
  extra?: (page: Page, width: number, theme: Theme) => Promise<void>,
): Promise<void> {
  for (const width of widths) {
    for (const theme of themes) {
      // A tall viewport keeps the whole hero above the fixed tab bar of a phone, so the bar covers no part of the shot.
      const { context, page } = await fresh(browser, width, { height: 1600 });
      await open(page, route, theme);
      if ((await page.locator(CANVAS).count()) > 0) await sceneReady(page);
      else await page.waitForTimeout(1500);
      await page.locator(HERO).first().screenshot({ path: join(dir(id), `${width}-${theme}.png`) });
      if (extra) await extra(page, width, theme);
      await context.close();
    }
  }
}

    const rows = [];
    for (const theme of THEMES) {
      const { context, page } = await fresh(browser, 375, { scale: 2, reduced: true });
      // The page itself takes both frames: at the first state change of the canvas (the entry) it sets two timers, at
      // 100 ms and at 2500 ms, and each timer reads the canvas pixels on the band color of the hero.
      await page.addInitScript(() => {
        const w = window as unknown as Record<string, unknown>;
        const shots: { at: number; entry: number; state: string; data: string; png: string }[] = [];
        w.reducedShots = shots;
        let entry: number | null = null;
        const grab = (canvas: HTMLCanvasElement) => {
          const hero = canvas.closest("section") as HTMLElement;
          const copy = document.createElement("canvas");
          copy.width = canvas.width;
          copy.height = canvas.height;
          const ctx = copy.getContext("2d")!;
          ctx.fillStyle = getComputedStyle(hero).backgroundColor;
          ctx.fillRect(0, 0, copy.width, copy.height);
          ctx.drawImage(canvas, 0, 0);
          shots.push({ at: performance.now(), entry: entry!, state: canvas.dataset.sceneState ?? "", data: canvas.toDataURL(), png: copy.toDataURL("image/png") });
        };
        new MutationObserver((changes) => {
          for (const change of changes) {
            const canvas = change.target as HTMLCanvasElement;
            if (entry === null && canvas.dataset.sceneState) {
              entry = performance.now();
              setTimeout(() => grab(canvas), 100);
              setTimeout(() => grab(canvas), 2500);
            }
          }
        }).observe(document, { subtree: true, attributeFilter: ["data-scene-state"] });
      });
      await open(page, WEEK, theme);
      await expect.poll(() => page.evaluate(() => ((window as unknown as { reducedShots: unknown[] }).reducedShots).length), { timeout: 10000 }).toBe(2);
      const shots = await page.evaluate(() => (window as unknown as { reducedShots: { at: number; entry: number; state: string; data: string; png: string }[] }).reducedShots);
      shots.forEach((shot, index) => writeFileSync(join(dir("AT-31"), `frame-${index === 0 ? "0.1" : "2.5"}s-375-${theme}.png`), Buffer.from(shot.png.split(",")[1], "base64")));
      rows.push({
        theme,
        states: (await statesOf(page)).map((s) => ({ state: s.state, ms_after_navigation: Math.round(s.at) })),
        entry_ms_after_navigation: Math.round(shots[0].entry),
        frames: shots.map((shot) => ({ ms_after_entry: Math.round(shot.at - shot.entry), state: shot.state })),
        framesEqual: shots[0].data === shots[1].data,
      });
      await context.close();
    }
    save("AT-31", "frames.json", rows);
  });

/* The analyzer: trade A of grade-clarity on the local API. */
async function gradeTradeA(page: Page): Promise<unknown> {
  let answer: unknown = null;
  page.on("response", async (response) => {
    if (response.url().startsWith("http://localhost:8080") && response.request().method() === "POST" && /analy/.test(response.url())) {
      try {
        answer = await response.json();
      } catch {
        // A stream or an empty body has no JSON.
      }
    }
  });
  await page.locator("input[type=password]").fill("localtest");
  await page.locator("button[type=submit]").click();
  await page.locator('button[aria-label="Pick Your Team"]').waitFor();
  await page.locator('button[aria-label="Pick Your Team"]').click();
  await page.locator('ul[aria-label="Teams"] button:has-text("behold the lamb")').click();
  await page.locator('button[aria-label="Pick Team Two"]').click();
  await page.locator('ul[aria-label="Teams"] button:has-text("Bo Girds")').click();
  await page.locator('button:has-text("Add a player Your Team gets")').click();
  await page.locator('ul[aria-label^="Players of"]').waitFor();
  await page.locator('ul[aria-label^="Players of"] button:has-text("Parker Washington")').click();
  await page.locator('button:has-text("Done")').click();
  await page.locator('button:has-text("Add a player Team Two gets")').click();
  await page.locator('ul[aria-label^="Players of"]').waitFor();
  await page.locator('ul[aria-label^="Players of"] button:has-text("Omar Cooper")').click();
  await page.locator('button:has-text("Done")').click();
  await page.locator('button:has-text("Grade the Trade")').click();
  await page.locator("text=Suggested Counters").waitFor({ timeout: 180000 });
  return () => answer;
}

          const box = section.getBoundingClientRect();
          const slots = [...section.querySelectorAll("[data-hero-slot]")];
          const eyebrow = section.querySelector('[data-hero-slot="eyebrow"]') ?? slots[0];
          const content = section.querySelector(":scope > div.relative")!;
          const all = [...content.querySelectorAll("*")].filter((el) => {
            const r = el.getBoundingClientRect();
            return r.height > 0 && r.width > 0;
          });
          const lowest = Math.max(...all.map((el) => el.getBoundingClientRect().bottom));
          const lowestEl = all.find((el) => el.getBoundingClientRect().bottom === lowest)!;
          return {
            label: section.getAttribute("aria-label"),
            slots: slots.map((s) => s.getAttribute("data-hero-slot")),
            spaceAboveEyebrowPx: Math.round(eyebrow.getBoundingClientRect().top - box.top),
            spaceUnderLowestContentPx: Math.round(box.bottom - lowest),
            lowestContentText: (lowestEl as HTMLElement).innerText?.replace(/\s+/g, " ").slice(0, 120),
            contentBottomToHeroBottom: Math.round(box.bottom - content.getBoundingClientRect().bottom),
            heroPadding: [getComputedStyle(section).paddingTop, getComputedStyle(section).paddingBottom],
            // Where the scene sits, and the count of scenes on the whole page after the verdict (AT-35).
            sceneInsideSlot: section.querySelector("canvas[data-hero-scene]")?.closest("[data-hero-slot]")?.getAttribute("data-hero-slot") ?? null,
            sceneTopPx: Math.round((section.querySelector("canvas[data-hero-scene]")?.getBoundingClientRect().top ?? box.top) - box.top),
            canvasesOnPage: document.querySelectorAll("canvas[data-hero-scene]").length,
            firstSlotTopPx: Math.round(slots[0].getBoundingClientRect().top - box.top),
          };
        });

        if (width === 375 && theme === "dark") save("AT-9", "tradeA.response.json", answer());
        await context.close();
      }
    }

#!/bin/zsh
# The evidence API of round 2: the real app of app_from_env with the Shortlist tap, from a detached worktree of 2e9f0f85.
source ~/.secrets
cd <scratch>/r2/dw/apps/api
export CORNERSTONE_RATE_LIMIT_PER_MINUTE=600 CORNERSTONE_ALLOWED_ORIGINS=http://127.0.0.1:8392 CORNERSTONE_REPO=<scratch>/r2/dw
export CORNERSTONE_RESULT_CACHE_DIR=<scratch>/r2/api-run/results CORNERSTONE_EGGS_DIR=<scratch>/r2/api-run/eggs CORNERSTONE_ASK_DIR=<scratch>/r2/api-run/ask
export SHORTLIST_OUT=<scratch>/r2/presses PYTHONPATH=<scratch>/r2
for slug in $(ls <scratch>/r2/dw/clients); do U=${(U)slug}; export CORNERSTONE_PASSPHRASE_$U=localtest CORNERSTONE_COMMISSIONER_KEY_$U=localkey; done
echo "API start $(date '+%Y-%m-%d %H:%M:%S')" > <scratch>/r2/api-start.txt
exec <scratch>/r2/dw/.venv/bin/uvicorn --factory evidence_tap:app --host 127.0.0.1 --port 8391

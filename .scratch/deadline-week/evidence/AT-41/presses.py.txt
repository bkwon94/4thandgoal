"""Split an API log into find presses, and name the players and teams of each Shortlist offer, gain drop, and trade."""
import json, re, sys
log, teams_file, streams = sys.argv[1], sys.argv[2], sys.argv[3:]
teams = json.load(open(teams_file))["teams"]
team = {t["roster_id"]: t["team"] for t in teams}
player = {p["player_id"]: p["name"] for t in teams for p in t["players"]}
def offer(code):
    # "6:8130+9997-11:3294" -> sides
    out = []
    for side in code.split("-"):
        rid, ids = side.split(":")
        out.append(f"{team[int(rid)]} gives " + " + ".join(player.get(i, i) for i in ids.split("+")))
    return "; ".join(out)
presses, cur = [], []
for line in open(log):
    if "httpx" in line or "typesafe_sdk" in line or "trade_news" in line:
        continue
    cur.append(line.rstrip())
    if "Press of find" in line:
        presses.append(cur); cur = []
for n, lines in enumerate(presses):
    print(f"===== press {n} (in log order) =====")
    for l in lines:
        m = re.search(r"Finder (shortlist|gain drop) (\S+): (.*)", l)
        if m:
            print(f"{m.group(1)}: {offer(m.group(2).rstrip(':'))}  [{m.group(2)}] {m.group(3)}")
        elif any(k in l for k in ("Spares tree", "Jev wave", "Press of", "retry", "Partner filter", "call:", "Traceback", "ERROR", "WARNING")):
            print(l.replace("INFO:     ", ""))
for s in streams:
    for l in open(s):
        if '"data-answer"' in l:
            d = json.loads(l.split("data: ", 1)[1])["data"]["document"]
            print(f"===== trades of {s.split('/')[-1]} =====")
            for t in d["trades"]:
                sides = "; ".join(f"{x.get('team', x['roster_id'])} gives " + " + ".join(p.get("name", str(p)) for p in x.get("gives", x.get("players", []))) for x in t["sides"])
                print(f"- {t.get('lean')}: {sides} | offer_id={t.get('offer_id')}")

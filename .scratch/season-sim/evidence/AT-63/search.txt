/* `random` gives a number in [0, 1), as Math.random does. The line from 0 to 1 holds one piece per run, as long as
   its share of the weights, and the run whose piece holds the number comes up. A run of weight 0 has no piece. */
export function pickRun(weights: number[], random: () => number): number {
  const total = weights.reduce((sum, weight) => sum + weight, 0);
  const at = random() * total;
  let end = 0;
  let last = 0;
  for (const [index, weight] of weights.entries()) {
    if (weight <= 0) continue;
    end += weight;
    last = index;
    if (at < end) return index;
  }
  // The sum of floats can end a hair under `at`, so the last run with a weight takes the rest of the line.
  return last;
}

  /* A new pulse enters from one end of a line. Half of them take a line near a weight, so the light gathers where the
     sides are. The first pulses of a wait start part way along their lines, so the wait does not open with a wave
     from the edges. */
  function spawn(nodes: MeshNodes, spread: boolean): Pulse {
    const dir: "h" | "v" = Math.random() < 0.5 ? "h" : "v";
    const sign: 1 | -1 = Math.random() < 0.5 ? 1 : -1;
    const lines = dir === "h" ? nodes.rows : nodes.cols;
    const length = dir === "h" ? nodes.cols : nodes.rows;

/* A small seeded random source (mulberry32), so a scatter is the same for the same seed and a test can check it. */
function randomOf(seed: number): () => number {
  let state = seed >>> 0;
  return () => {
    state = (state + 0x6d2b79f5) >>> 0;
    let t = state;
    t = Math.imul(t ^ (t >>> 15), t | 1);
    t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}

/* The start place of each light: a scatter inside the padding of the box. */
export function scatterOf(count: number, box: SceneBox, seed: number): ScenePoint[] {
  const random = randomOf(seed);
  return Array.from({ length: count }, () => ({
    x: box.padding + random() * (box.width - 2 * box.padding),
    y: box.padding + random() * (box.height - 2 * box.padding),
  }));
}

/* The faint dust of the scene: 40 to 80 tiny points (PRD). Each drifts slowly in its own direction. */
export function dustOf(count: number, box: SceneBox, seed: number): DustPoint[] {
  const random = randomOf(seed);
  return Array.from({ length: Math.max(40, Math.min(80, count)) }, () => ({
    x: random() * box.width,
    y: random() * box.height,
    vx: (random() - 0.5) * 16,
    vy: (random() - 0.5) * 10,
  }));
}

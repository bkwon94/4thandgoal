const say = (line) => { log.push(line); console.log(line); };

async function open(browserType, width, height, theme = "light", options = {}) {
  const browser = await browserType.launch();
  const page = await browser.newPage({ viewport: { width, height }, ...options });
  page.setDefaultTimeout(90000);
  await page.addInitScript((t) => localStorage.setItem("cpc.theme", t), theme);
  await page.goto(URL);
  await page.locator("text=Which team is yours?").waitFor();
  await page.waitForTimeout(1200);
  return { browser, page };
}
async function pickTeam(page, team) {
  await page.locator(`[role=dialog] button:has-text("${team}")`).first().click();
  await page.locator('[aria-label="Suggestions"]').waitFor();
  await page.waitForTimeout(1500);
}
async function passphraseThen(page, team) {
  await page.locator(`[role=dialog] button[aria-pressed]:not(:has-text("${team}"))`).first().click();
  await page.locator('input[aria-label="Ask a question"]').waitFor();
  await page.fill('input[aria-label="Ask a question"]', "Week recap");
  await page.click('button[aria-label="Send"]');
  await page.fill("input[type=password]", "localtest");
  await page.click('button[aria-label="Ask with this passphrase"]');
  await page.waitForTimeout(2500);
  await page.locator('button[aria-label^="Your team:"]').first().click();
  await page.waitForTimeout(1000);
  await pickTeam(page, team);
}
async function more(page, name) {
  await page.locator('[aria-label="Suggestions"] > button:has-text("More")').click();
  const row = page.locator(`ul[aria-label="All Topics"] button:enabled[aria-label^="${name}:"]`);
  await row.waitFor();
  await page.waitForTimeout(500);
  await row.click();
}

  const target = page.locator('li[data-slot] > button[aria-expanded="false"]').first();

  // The blur of the detail at 3 moments after the tap, read from the computed filter of the opened panel.

  for (const ms of [30, 120, 400]) {

"none"; }, slotIndex));
  }
  await page.waitForTimeout(1200);
  const after = await tops();
  say(`# AT-9: Teehee (cpc roster 12), Start or sit, 375 x 900, light, at ${new Date().toISOString()}. Tapped row index ${slotIndex}.`);

// The scroll pass of the runner: down one half screen at a time to the end and back, so each reveal on scroll plays.
async function scrollPass(page) {
  await page.evaluate(async () => {
    const pause = () => new Promise((done) => setTimeout(done, 50));
    const start = window.scrollY;
    const step = Math.max(200, Math.floor(window.innerHeight / 2));
    for (let y = 0; y < document.documentElement.scrollHeight; y += step) { window.scrollTo(0, y); await pause(); }
    window.scrollTo(0, document.documentElement.scrollHeight); await pause();
    window.scrollTo(0, start);
  });
  await page.waitForTimeout(1200);
}
// Waits until the count of buttons in the Suggestions row stays the same for 1 s, so the chips of the old picks are gone.
async function stableRow(page) {
  let last = -1, since = Date.now();
  for (;;) {
    const count = await page.locator('[aria-label="Suggestions"] button').count();
    if (count !== last) { last = count; since = Date.now(); }
    if (Date.now() - since >= 1000) return count;
    await page.waitForTimeout(100);
  }
}
// The box of the row and of each button, and the pad and margin of the row from its computed style.
const measure = (page) => page.$eval('[aria-label="Suggestions"]', (row) => {
  const r = row.getBoundingClientRect(); const s = getComputedStyle(row);
  const px = (v) => parseFloat(v);
  const chips = [...row.querySelectorAll("button")].map((b) => { const c = b.getBoundingClientRect(); return `text "${b.textContent.trim()}" aria-label "${b.getAttribute("aria-label") ?? "none"}"@top${c.top.toFixed(1)}/h${c.height.toFixed(1)}/w${Math.round(c.width)}`; });
  const layout = r.height + px(s.marginTop) + px(s.marginBottom);
  return `row box height ${r.height.toFixed(1)} px (top ${r.top.toFixed(1)}); padding top ${s.paddingTop}, bottom ${s.paddingBottom}; margin top ${s.marginTop}, bottom ${s.marginBottom}; layout height (box plus margins) ${layout.toFixed(1)} px; ${chips.length} buttons: ${chips.join("; ")}`;
});

if (which === "AT-51") {
  // An iPhone 15 size in WebKit. Playwright cannot open the iOS keyboard, so the run sets its height: on the focus of
  // the passphrase box it shrinks the viewport by 336 px, the height of the iOS keyboard with its bar on that phone.
  const KEYBOARD = 336;
  const { browser, page } = await open(webkit, 393, 852, "light", { hasTouch: true, isMobile: true, deviceScaleFactor: 3 });
  await pickTeam(page, "Bo Girds");
  await more(page, "Week recap");
  await page.locator('[data-block="recap"]').waitFor();
  await page.waitForTimeout(1500);
  await page.fill('input[aria-label="Ask a question"]', "How did my bench do?");
  await page.click('button[aria-label="Send"]');
  await page.locator("input[type=password]").waitFor();
  await page.waitForTimeout(1500);
  if (process.argv[3] === "tap") {
    // Variant: the keyboard height is set first, then a tap on the box focuses it. The tap scrolls the box into view,
    // as iOS scrolls a focused field into the visual viewport.
    await page.setViewportSize({ width: 393, height: 852 - KEYBOARD });
    await page.waitForTimeout(800);
    await page.locator("input[type=password]").tap();
  } else {
    await page.focus("input[type=password]");
    await page.setViewportSize({ width: 393, height: 852 - KEYBOARD });
  }
  await page.waitForTimeout(1500);
  const read = await page.evaluate(() => {
    const field = document.querySelector("input[type=password]");
    const box = field.closest("form") ?? field.parentElement;
    const ask = document.querySelector('input[aria-label="Ask a question"]');
    let foot = ask; while (foot && getComputedStyle(foot).position !== "fixed" && getComputedStyle(foot).position !== "sticky") foot = foot.parentElement;
    const r = (el) => { const b = el.getBoundingClientRect(); return { top: Math.round(b.top), bottom: Math.round(b.bottom) }; };
    return { viewport: window.innerHeight, visual: window.visualViewport?.height, box: r(box), foot: foot ? r(foot) : null, focused: document.activeElement === field, keyboardAttr: document.documentElement.hasAttribute("data-keyboard") };
  });
  say(`# AT-51: WebKit (Playwright webkit), 393 x 852, mobile, touch, light, at ${new Date().toISOString()}. Bo Girds of cpc with the Week recap Answer in the Thread, then a typed question that shows the passphrase box.`);
  say(process.argv[3] === "tap" ? `# The run sets the keyboard height first (the viewport goes from 852 to ${852 - KEYBOARD} px tall), then taps the passphrase box.` : `# The run focuses the passphrase box, then sets the keyboard height: the viewport goes from 852 to ${852 - KEYBOARD} px tall.`);
  say(JSON.stringify(read));
  say(`box above the foot: ${read.foot ? read.box.bottom <= read.foot.top : "no fixed foot found"}; box in view: ${read.box.top >= 0 && read.box.bottom <= read.viewport}`);
  await page.screenshot({ path: `${E}/AT-51/${process.argv[3] === "tap" ? "resize-then-tap" : "focus-then-resize"}-393-light.png` });
  writeFileSync(`${E}/AT-51/positions-${process.argv[3] === "tap" ? "resize-then-tap" : "focus-then-resize"}.txt`, log.join("\n") + "\n");
  await browser.close();
}

import json
P=json.load(open("players.json"))
for n in ["cpc","norcal"]:
    m=json.load(open(f"m-{n}.json")); u={x["user_id"]:(x.get("metadata",{}).get("team_name") or x["display_name"]) for x in json.load(open(f"u-{n}.json"))}
    r={x["roster_id"]:u.get(x["owner_id"],"?") for x in json.load(open(f"r-{n}.json"))}
    slots=[s for s in json.load(open(f"l-{n}.json"))["roster_positions"] if s not in ("BN","IR","TAXI")]
    for x in sorted(m,key=lambda x:x["roster_id"]):
        st=x.get("starters") or []
        out=[];flags=[]
        for s,pid in zip(slots, st+["0"]*(len(slots)-len(st))):
            if pid=="0": out.append(f"{s}:EMPTY"); flags.append("E"); continue
            p=P.get(pid,{}); stt=p.get("injury_status")
            out.append(f"{s}:{p.get('full_name',pid)}({p.get('team')},{stt})")
            if stt: flags.append(stt)
        print(n,x["roster_id"],r[x["roster_id"]],flags)
        print("   "," | ".join(out))

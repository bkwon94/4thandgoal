"""
Rebuild topics.json of a saved week in the published folder, from the saved raw week only.
It never calls Sleeper: the settings loader reads the saved league file and raises if it is missing.
"""
import sys
from pathlib import Path

from cornerstone.pipeline import _saved_league_week
from cornerstone.topics import write_topics
from cornerstone.value_models import VALUE_MODELS
from league.client import load_client
from league.provider.raw import RawLeague
from league.provider.sleeper import league_settings

root, slug, season, week = Path(sys.argv[1]), sys.argv[2], int(sys.argv[3]), int(sys.argv[4])


def saved_settings(league_id, cache):
    if not cache.exists():
        raise SystemExit(f"no saved league file at {cache}; refusing to fetch")
    return league_settings(RawLeague.model_validate_json(cache.read_text(encoding="utf-8")))


client = load_client(root, slug, season, saved_settings, VALUE_MODELS)
league_week = _saved_league_week(client, season, week)
print(f"client={slug} folder={client.folder} league_week={'parsed' if league_week else 'None'}")
out = write_topics(client.folder / "published", season, week, league_week)
print(f"wrote {out}")

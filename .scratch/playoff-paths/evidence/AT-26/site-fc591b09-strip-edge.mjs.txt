
import { createRequire } from "node:module";
const require = createRequire(process.env.SITE_APP + "/package.json");
const { webkit, devices } = require("@playwright/test");
const [w, h, theme] = process.argv.slice(2);
const browser = await webkit.launch();
const ctx = await browser.newContext({ ...devices[+w < 500 ? "iPhone 15" : "iPad Pro 11"], viewport: { width: +w, height: +h }, reducedMotion: "no-preference" });
await ctx.addInitScript((theme) => { localStorage.setItem("cpc.theme", theme); localStorage.setItem("cpc.your-team", "6"); localStorage.setItem("cpc.tour-seen", "1"); localStorage.setItem("cpc.passphrase", "localtest"); localStorage.setItem("cpc.announcement.the-race-2026", "seen"); }, theme);
const page = await ctx.newPage();
await page.goto("http://127.0.0.1:4391/", { waitUntil: "networkidle" });
await page.waitForTimeout(1000);
await page.locator("button", { hasText: "More" }).first().click();
const chip = page.locator("button[aria-label^='Playoff race']", { hasText: "Playoff race" }).first();
await chip.waitFor({ state: "visible" }); await page.waitForLoadState("networkidle"); await page.waitForTimeout(800);
await chip.click();
await page.locator("a", { hasText: "Open The Race" }).first().waitFor({ state: "visible", timeout: 20000 });
await page.waitForTimeout(1500);
const top = await page.evaluate(() => { const b = [...document.querySelectorAll("*")].find((e) => e.children.length === 0 && e.textContent.trim() === "Playoff race"); return b ? b.getBoundingClientRect().top + scrollY - 80 : 0; });
await page.evaluate((y) => window.scrollTo(0, y), top); await page.waitForTimeout(600);
const r = await page.evaluate(() => {
  const hdr = document.querySelector('header[aria-label="Thread header"]');
  const leaf = [...document.querySelectorAll("*")].find((e) => e.children.length === 0 && e.textContent.trim() === "Playoff race");
  let bub = leaf; for (let i = 0; i < 4 && bub; i++) { const cs = getComputedStyle(bub); if (cs.backgroundColor !== "rgba(0, 0, 0, 0)") break; bub = bub.parentElement; }
  const R = (e) => { const b = e.getBoundingClientRect(); return { top: +b.top.toFixed(1), bottom: +b.bottom.toFixed(1), left: +b.left.toFixed(1), right: +b.right.toFixed(1) }; };
  return { scrollY, header: R(hdr), headerBg: getComputedStyle(hdr).backgroundColor, bubble: R(bub), bubbleBg: getComputedStyle(bub).backgroundColor, bubbleBorder: getComputedStyle(bub).borderBottom, bubbleShadow: getComputedStyle(bub).boxShadow };
});
console.log(JSON.stringify(r));
await page.screenshot({ path: process.env.OUT, clip: { x: 0, y: 0, width: +w, height: 260 } });
await browser.close();

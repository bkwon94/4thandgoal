# The eggs store (the eggs package, not the front door), data/eggs/cpc.sqlite of the repo, which the API also opens:
$ sqlite3 data/eggs/cpc.sqlite .schema
CREATE TABLE eggs (
    id INTEGER PRIMARY KEY,
    week INTEGER NOT NULL,
    page TEXT NOT NULL,
    target TEXT NOT NULL,
    action TEXT NOT NULL,
    conditions TEXT NOT NULL,
    letter TEXT NOT NULL,
    dropped_at TEXT NOT NULL
);
CREATE TABLE claims (
    egg_id INTEGER NOT NULL REFERENCES eggs (id),
    roster_id INTEGER NOT NULL,
    word TEXT,
    claimed_at TEXT NOT NULL,
    browser TEXT,
    undone_at TEXT
);
CREATE UNIQUE INDEX claims_egg_id ON claims (egg_id) WHERE undone_at IS NULL;

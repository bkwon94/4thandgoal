# slice.py:
"""
Print the weak spots slice and data block of one CPC roster from the live league, with the real code of the API.
It calls no Claude and no Jev. The slice is the input that Claude reads; the block is the data block of the stream.
"""
import json
import sys

from cornerstone_api.app import app_from_env
from cornerstone_api.personal_topics import LiveCache, _load

app = app_from_env()
live = _load(app.state.served["cpc"], 2026, LiveCache(60))
facts = live.facts("weak_spots", int(sys.argv[1]))
print(json.dumps({"roster_id": int(sys.argv[1]), "slice": facts.slice, "block": facts.block}, indent=2))

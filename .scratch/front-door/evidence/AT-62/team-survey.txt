# holes.py:
"""
Print the weak spots data block of each CPC roster from the live league, with the real code of the API.
It calls no Claude and no Jev: it reads only the facts that the /personal stream sends as its first data part.
"""
import json

from cornerstone_api.app import app_from_env
from cornerstone_api.personal_topics import LiveCache, _load

app = app_from_env()
served = app.state.served["cpc"]
live = _load(served, 2026, LiveCache(60))
print(f"Live league: week {live.snapshot.week}")
for rid in sorted(live.snapshot.rosters):
    facts = live.facts("weak_spots", rid)
    block = facts.block
    holes = [s for s in facts.slice["starters"] if s["hole"]]
    print(f"\nroster {rid}, {live.snapshot.teams[rid].name}: {block['fact']}; stats {[(s['label'], s['value']) for s in block['stats']]}")
    for row in block["rows"]:
        print(f"  row {row['label']}: {row['value']} · {row['note']}")
    print(f"  starters with hole=true: {[(s['slot'], s['player'], s['can_play_now']) for s in holes]}")

/**
 * The iPhone WebKit shots of each trade-block screenshot test at 390x660, and the WebKit runs of AT-24, AT-26, AT-27,
 * and AT-32, on the real local site and the real local API. The iPhone shots reuse the step list of each test.
 */
import { devices, test } from "@playwright/test";
import { EVID, captureFind, listFrom, runEvidence } from "./zz-tb-shared";
import { blockedFlow, fitFlow, keyFlow, motionOffFlow, reloadFlow } from "./zz-tb-flows";
import { join } from "node:path";

const { defaultBrowserType: _p, ...IPHONE } = devices["iPhone 15"];
const { defaultBrowserType: _t, ...IPAD } = devices["iPad Pro 11"];
const PHONE = { ...IPHONE, viewport: { width: 390, height: 660 } };
const LABEL = "iphone-390x660";
const IDS = listFrom(process.env.TB_IDS, "AT-10 AT-10/walker AT-11 AT-12 AT-13 AT-14 AT-15 AT-16 AT-17 AT-19 AT-20 AT-21 AT-22 AT-31");
const FLOWS = listFrom(process.env.TB_FLOWS, "AT-18 AT-24 AT-26 AT-27 AT-32");

test.describe("iPhone 15 WebKit 390x660", () => {
  test.use({ ...PHONE, reducedMotion: "no-preference" });
  for (const id of IDS) {
    test(`${id} shots`, async ({ page }) => {
      captureFind(page, join(EVID, id), LABEL);
      await runEvidence(page, id, LABEL, "light");
    });
  }
  if (FLOWS.includes("AT-18")) test("AT-18 blocked", async ({ page }) => blockedFlow(page, LABEL, "light"));
  if (FLOWS.includes("AT-32")) test("AT-32 reload", async ({ page }) => reloadFlow(page, LABEL, "light"));
  if (FLOWS.includes("AT-26")) test("AT-26 keys", async ({ page }) => keyFlow(page, LABEL, "light"));
  if (FLOWS.includes("AT-27")) for (const theme of ["light", "dark"] as const) test(`AT-27 fit ${theme}`, async ({ page }) => fitFlow(page, LABEL, theme, 660));
});

test.describe("iPhone 15 WebKit 390x660, motion off", () => {
  test.use({ ...PHONE, reducedMotion: "reduce" });
  if (FLOWS.includes("AT-24")) test("AT-24 motion off", async ({ page }) => motionOffFlow(page, LABEL));
});

for (const [label, viewport] of [["ipad-810x980", { width: 810, height: 980 }], ["ipad-1080x740", { width: 1080, height: 740 }]] as const) {
  test.describe(`iPad WebKit ${label}`, () => {
    test.use({ ...IPAD, viewport, reducedMotion: "no-preference" });
    if (FLOWS.includes("AT-27")) for (const theme of ["light", "dark"] as const) test(`AT-27 fit ${label} ${theme}`, async ({ page }) => fitFlow(page, label, theme, viewport.height));
  });
}

# lockdiff.py
import subprocess,re,collections
def keys(ref):
    txt=subprocess.run(["git","show",f"{ref}:pnpm-lock.yaml"],capture_output=True,text=True).stdout
    sec=None; out={'packages':set(),'snapshots':set()}; head=[]
    for l in txt.splitlines():
        m=re.match(r'^(\S[^:]*):\s*$',l)
        if m: sec=m.group(1); continue
        if sec in out:
            m=re.match(r"^  '?([^' ].*?)'?:\s*(\{\})?$",l)
            if m: out[sec].add(m.group(1))
        if l.startswith('lockfileVersion') or l.startswith('settings') : head.append(l)
    return out,txt
def split(k):
    # name@version, scoped names start with @
    i=k.rfind('@',1) if not k.startswith('@') else k.find('@',1)
    return k[:i],k[i+1:]
a,ta=keys("main"); b,tb=keys("HEAD")
print("lockfileVersion main:", [l for l in ta.splitlines() if l.startswith('lockfileVersion')], " spec:", [l for l in tb.splitlines() if l.startswith('lockfileVersion')])
for sec in ['packages']:
    A,B=a[sec],b[sec]
    print(f"\n[{sec}] entries: main {len(A)}, spec {len(B)}; removed {len(A-B)}, added {len(B-A)}, same {len(A&B)}")
    va=collections.defaultdict(set); vb=collections.defaultdict(set)
    for k in A: n,v=split(k); va[n].add(v)
    for k in B: n,v=split(k); vb[n].add(v)
    print("\nAdded entries (name@version on the Spec branch, not on main):")
    for k in sorted(B-A):
        n,v=split(k); note=" (name also on main at "+", ".join(sorted(va[n]))+")" if n in va else " (new name)"
        print("  +",k,note)
    print("\nRemoved entries (on main, not on the Spec branch):")
    for k in sorted(A-B):
        n,v=split(k); note=" (name stays on the Spec branch at "+", ".join(sorted(vb[n]))+")" if n in vb else ""
        print("  -",k,note)
    print("\nNames on both branches whose set of locked versions differs:")
    c=0
    for n in sorted(set(va)&set(vb)):
        if va[n]!=vb[n]: c+=1; print(f"  {n}: main {sorted(va[n])} -> spec {sorted(vb[n])}; versions of main kept: {sorted(va[n]&vb[n])}")
    print("  count:",c)
    lost=[k for k in A-B if split(k)[0] in vb]
    print("\nEntries of main whose name stays but whose version is gone:",lost or "none")

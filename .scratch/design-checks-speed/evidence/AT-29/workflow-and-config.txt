  # The unit half needs no browser. "make check-site" runs both halves in sequence.
  # The unit half runs in a matrix of shards, and each job runs bun test on the files of its shard only, split by the time
  # of each file in apps/site/bun-test-timings.json. With --parallel, the timings also start the slowest files first. Each
  # job also runs the lint, the format check, the client build, and typecheck. The measured times of ticket 15 of
  # design-checks-speed set the count. With 4 workers, the count of an ubuntu-latest runner, the slowest shard takes 47 s
  # with 1 shard and 34 s with 2, 3, or 4, because one test file takes 33 s alone. One more job costs about 42 s of setup,
  # so a second shard saves less than it costs. With 1 worker, the slowest shard takes 170 s with 1 shard and 97 s with 2.
  site-unit:
    runs-on: ubuntu-latest
    timeout-minutes: 15
    strategy:
      # Each shard runs to its end, so a red run shows the failures of every shard, not only of the first.
      fail-fast: false
      matrix:
        shard: [1]
    env:
      # scripts/bun-test.mjs reads it, such as "1/1". The total comes from the matrix, so the list above is the one count.
      BUN_TEST_SHARD: ${{ matrix.shard }}/${{ strategy.job-total }}
